问题单调才能用滑窗
```go
sum := 0
l := 0
for r := 0; r < n; r++ {
	更新窗口（sum += nums[r]）
	for 满足条件可以收缩(l <= r && 条件) {
		sum -= nums[l]
		l++
	}
	if 可以更新答案 {
		更新答案（minLength = r - l + 1）
	}
}
// 窗口[l, r]
```

双指针
```go
func maxArea(height []int) int {
    l, r := 0, len(height)-1
    res := 0
    for l <= r {
        res = max(res, (r-l)*min(height[l], height[r]))
        if height[l] < height[r] {
            l++
        } else {
            r--
        }
    }
    return res
}
```
